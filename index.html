-- Them cot luu thong tin tai khoan da giao trong don hang ALTER TABLE public.orders ADD COLUMN IF NOT EXISTS account_login text, ADD COLUMN IF NOT EXISTS account_password text; -- Khach chi xem duoc don hang cua chinh minh ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Customer read own orders" ON public.orders; CREATE POLICY "Customer read own orders" ON public.orders FOR SELECT TO authenticated USING (user_id = auth.uid()); -- Ham mua tai khoan: tru tien, tao don, giao tai khoan CREATE OR REPLACE FUNCTION public.purchase_product(p_product_id bigint) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_user_id uuid := auth.uid(); v_price bigint; v_title text; v_balance bigint; v_login text; v_password text; v_order_id text; BEGIN IF v_user_id IS NULL THEN RAISE EXCEPTION 'Vui long dang nhap truoc khi mua.'; END IF; SELECT title, price::bigint INTO v_title, v_price FROM public.products WHERE id = p_product_id AND is_available = true FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Tai khoan khong con hang.'; END IF; SELECT balance INTO v_balance FROM public.wallets WHERE user_id = v_user_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Khong tim thay vi tien.'; END IF; IF v_balance < v_price THEN RAISE EXCEPTION 'So du vi khong du. Hay nap them tien.'; END IF; SELECT account_login, account_password INTO v_login, v_password FROM public.product_credentials WHERE product_id = p_product_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'San pham chua co thong tin tai khoan.'; END IF; UPDATE public.wallets SET balance = balance - v_price WHERE user_id = v_user_id; UPDATE public.products SET is_available = false WHERE id = p_product_id; INSERT INTO public.orders (user_id, product_id, amount, status, account_login, account_password) VALUES (v_user_id, p_product_id, v_price, 'completed', v_login, v_password) RETURNING id::text INTO v_order_id; INSERT INTO public.wallet_transactions (user_id, amount, transaction_type, description) VALUES (v_user_id, v_price, 'purchase', 'Mua ' || COALESCE(v_title, 'tai khoan') || ' - don #' || v_order_id); RETURN jsonb_build_object( 'success', true, 'order_id', v_order_id, 'title', v_title, 'amount', v_price, 'account_login', v_login, 'account_password', v_password ); END; $$; REVOKE ALL ON FUNCTION public.purchase_product(bigint) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.purchase_product(bigint) TO authenticated; -- Ham Admin them san pham va thong tin tai khoan game CREATE OR REPLACE FUNCTION public.admin_add_product( p_title text, p_description text, p_price bigint, p_image_url text, p_account_login text, p_account_password text ) RETURNS bigint LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_product_id bigint; BEGIN IF auth.uid() IS NULL OR NOT public.is_admin() THEN RAISE EXCEPTION 'Ban khong co quyen Admin.'; END IF; IF COALESCE(trim(p_title), '') = '' OR p_price < 1000 OR COALESCE(trim(p_account_login), '') = '' OR COALESCE(p_account_password, '') = '' THEN RAISE EXCEPTION 'Thong tin san pham khong hop le.'; END IF; INSERT INTO public.products (title, description, price, image_url, is_available) VALUES (trim(p_title), p_description, p_price, NULLIF(trim(p_image_url), ''), true) RETURNING id INTO v_product_id; INSERT INTO public.product_credentials (product_id, account_login, account_password) VALUES (v_product_id, p_account_login, p_account_password); RETURN v_product_id; END; $$; REVOKE ALL ON FUNCTION public.admin_add_product( text, text, bigint, text, text, text ) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.admin_add_product( text, text, bigint, text, text, text ) TO authenticated;
